# BloomIndex backend — Python dependencies
# Install:   uv pip install --python .venv/Scripts/python.exe -r backend/requirements.txt
#            (or: pip install -r backend/requirements.txt)
# Tested on: Python 3.14 (trial) — 3.13 fallback if native crashes appear.
#   Venv:    uv venv --python 3.14 .venv  (repo root; .venv/ is git-ignored)
#   spaCy model out-of-band: ``python -m spacy download en_core_web_sm``
#
# Pin policy:
#   * Hard pins (==) for anything in the ML stack (numpy, transformers,
#     sentence-transformers, qdrant-client, langchain*) — these have
#     load-bearing API surfaces and any version drift quickly breaks
#     embedding shape, reranker tokenization, or vector-store schema.
#   * Hard pins for the FastAPI core too — pydantic v1↔v2, FastAPI's
#     dependency-injection internals, and uvicorn's event-loop hooks
#     move in lockstep.
#   * Everything else gets a pin because reproducibility > convenience.
#
# When bumping a pin: run ``pytest backend/tests/ -v`` AND do one full
# end-to-end PDF upload + query before committing. The embedding-dim
# clamp + Qdrant hybrid retrieval + cross-encoder reranker have
# several silent-breakage modes that unit tests don't catch.

# === FastAPI core ===========================================================
fastapi==0.141.1
uvicorn[standard]==0.52.4
# uvloop: libuv-backed asyncio event loop. uvicorn auto-detects + switches
# the policy when present — ~25-40% throughput gain on the streaming SSE +
# chunked upload paths. No Windows wheels exist, so the env marker keeps
# the install Linux/macOS-only; Windows falls back to the stdlib
# SelectorEventLoop with no code change.
uvloop==0.22.1; sys_platform != "win32"
pydantic==2.13.5
python-multipart==0.0.32

# === HTTP ===================================================================
httpx==0.28.1
# Official OpenAI SDK — the single LLM boundary (RAG, NER, RAGAS eval).
# Pinned directly; do not rely on transitive installs via ragas/instructor.
# v3 rides on httpx2 (separate fork) — our httpx==0.28.1 pin is unaffected.
# NOTE: instructor/langchain-openai declare openai<3; both import fine and
# the RAGAS handoff constructs, but re-run an eval row after any bump.
openai==3.13.0

# === NLP — dictionary matchers (spaCy PhraseMatcher) =======================
# spaCy model is downloaded out-of-band: ``python -m spacy download en_core_web_sm``
spacy==3.8.16

# === RAG: LangChain + Qdrant + embeddings + reranker =======================
langchain==1.4.0
langchain-community==0.4.2
langchain-core==1.6.3
langchain-huggingface==1.2.2
langchain-qdrant==1.1.0
langchain-experimental==0.4.2
langchain-text-splitters==1.1.2
# qdrant-client version MUST match the Qdrant Server image tag in
# scripts/qdrant.{sh,ps1} — mismatched protocol
# versions trigger gRPC schema errors on collection introspection.
qdrant-client==1.19.0
# Sparse BM25 encoder for hybrid retrieval. Loaded lazily via
# ``FastEmbedSparse(model_name="Qdrant/bm25")`` at first query.
# langchain-qdrant 1.x ships fastembed only as an *extras* dependency
# (``langchain-qdrant[fastembed]``), so a plain install does NOT
# pull it. Declaring it explicitly here keeps hybrid retrieval working
# on a clean install from any machine.
fastembed==0.8.0
sentence-transformers==6.0.1
# transformers is kept in lockstep with the sentence-transformers
# integration. Bumping one without the other typically breaks the
# cross-encoder reranker's tokenizer config.
transformers==5.17.0
# Permissive JSON parser used by the structured-output citation pass.
# Production LLMs (especially smaller models in JSON mode) emit common
# malformations: trailing commas, code-fenced output, single quotes,
# unclosed braces, smart quotes. ``json_repair`` recovers the intended
# object from these without an extra LLM round-trip.
json-repair==0.63.4

docling==2.126.0
beautifulsoup4==4.15.0

# nh3 0.3.x renamed several arguments — ``sanitize.py`` was audited

nh3==0.3.7
